import os
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

# The target URL you just found
url = httpsocwindia.comconsumer_cornerestimated_supply_map

# Directory to save the ESR maps
download_dir = ocw_esr_maps
os.makedirs(download_dir, exist_ok=True)

print(Fetching page...)
headers = {'User-Agent' 'Mozilla5.0 (Windows NT 10.0; Win64; x64)'}
response = requests.get(url, headers=headers)
soup = BeautifulSoup(response.text, 'html.parser')

# Extract all anchor tags ending in .pdf
pdf_links = set()
for link in soup.find_all('a', href=True)
    href = link.get('href')
    if href.lower().endswith('.pdf')
         pdf_links.add(urljoin(url, href))

print(fFound {len(pdf_links)} ESR maps. Downloading...)

# Download loop
for pdf_url in pdf_links
    # Clean up the filename
    filename = pdf_url.split('')[-1].replace('%20', '_')
    filepath = os.path.join(download_dir, filename)
    
    print(fSaving {filename}...)
    try
        pdf_res = requests.get(pdf_url, headers=headers)
        pdf_res.raise_for_status()
        with open(filepath, 'wb') as f
            f.write(pdf_res.content)
    except Exception as e
        print(fFailed to download {filename} {e})

print(All downloads complete. Q1·F1 is back on.)
